    \section*{Expressions conditionnelles}
    \subsection*{Syntaxe:}
    \noindent Un bloc indenté n'est exécuté que \textbf{si et seulement si} la condition dont il dépend est \textbf{vérifiée}.
    \begin{lstlisting}
    if <condition 1>:
        Instruction
        Instruction
        ...
    elif <condition 2>:
        Instruction
        Instruction
        ...
    else:
        Instruction
        Instruction
        ...
    \end{lstlisting}

    \section*{Les boucles}
    
    % La fonction range() génère la séquence sans la stocker entièrement en mémoire
    \subsection*{La fonction range():}
    \noindent La fonction \texttt{range()} permet de créer une séquence d'entier selon le modèle suivant:
    \begin{center}
    \begin{lstlisting}[language=Python]
    range(start*,stop,step*)
    \end{lstlisting} 
    \end{center}
    \begin{itemize}
        \item \texttt{start}: Valeur initiale de la séquence (par défaut 0)
        \item \texttt{stop}: Valeur limite d'arrêt (exclue)
        \item \texttt{step}: Valeur du pas de la séquence (par défaut 1)
        \item * paramètre optionnel
    \end{itemize}

    \subsubsection*{Exemples:}
    \begin{lstlisting}[language=Python]
    range(1,10,2)     # 1, 3, 5, 7, 9
    range(10, 2, -2)  # 10, 8, 6, 4
    range(2, 5)       # 2, 3, 4
    range(5)          # 0, 1, 2, 3, 4
    range(10, 5, -3)  # vide
    range(1, 10, -3)  # vide
    \end{lstlisting} 

    \subsection*{La boucle for:}
    \subsubsection*{Syntaxe:}
    \begin{lstlisting}
        for <variable> in <sequence>:
            Instruction
            Instruction
        \end{lstlisting} 
    <variable> prend successivement toutes les valeurs de <sequence> et exécute le bloc indenté pour chacune d'elles.
    \newpage

    \subsection*{La boucle while:}
    \subsubsection*{Syntaxe:}
    \begin{lstlisting}
    <Initialisation de la condition>
    while <condition>:
        Instruction
        Instruction
        ...
    \end{lstlisting} 
    Attention: Quelque part à l'intérieur du \texttt{while} la condition doit évoluer afin de ne pas créer une boucle infinie.

    \subsubsection*{Mettre fin à une boucle infinie:}
    \begin{itemize}
        \item Dans Thonny
        \begin{itemize}
            \item Cliquer sur le bouton rouge "Stop/Arrêter"
            \item Ou appuyer sur Ctrl + F2
        \end{itemize}
        \item Dans Visual Studio Code
        \begin{itemize}
            \item Cliquer sur le bouton rouge "Stop/Arrêter"
            \item Ou mettre le focus sur le terminal et appuyer sur Ctrl + C
        \end{itemize}
    \end{itemize}
